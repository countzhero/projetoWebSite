create database flowfit;
use flowfit;

-- Tabela de usuários
CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    senha VARCHAR(255) NOT NULL,
    perfil ENUM('aluno','instrutor','gestor') NOT NULL,
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);S


-- Tabela de equipamentos
CREATE TABLE equipamentos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    categoria VARCHAR(50),
    tempo_limite INT NOT NULL,
    status ENUM('Livre','Ocupado','Esgotado') DEFAULT 'Livre'
);

-- Tabela de utilização
CREATE TABLE utilizacao (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT,
    equipamento_id INT,
    inicio DATETIME,
    fim DATETIME,
    status ENUM('Ativo','Finalizado','Tempo Esgotado'),

    FOREIGN KEY(usuario_id)
    REFERENCES usuarios(id),

    FOREIGN KEY(equipamento_id)
    REFERENCES equipamentos(id)
);
